# 三级联动

> 省份-城市-区县三级联动下拉选择，选择省份后自动更新城市列表，选择城市后自动更新区县列表

## 一、数据结构

三个JS文件提供省市区数据，通过 `<script>` 引入：

```html
<script src="data/provinces.js"></script>
<script src="data/cities.js"></script>
<script src="data/areas.js"></script>
```

### 数据格式

```javascript
// provinces.js — 省份数据
let provinces = [
    { "code": "11", "name": "北京市" },
    { "code": "12", "name": "天津市" },
    { "code": "13", "name": "河北省" },
    // ...
];

// cities.js — 城市数据
let cities = [
    { "code": "1101", "name": "北京市", "provinceCode": "11" },
    { "code": "1301", "name": "石家庄市", "provinceCode": "13" },
    // ...
];

// areas.js — 区县数据
let areas = [
    { "code": "110101", "name": "东城区", "cityCode": "1101" },
    { "code": "110102", "name": "西城区", "cityCode": "1101" },
    // ...
];
```

### 数据关联关系

```
省份(provinces)  ──provinceCode──→  城市(cities)  ──cityCode──→  区县(areas)
   code: "11"                    provinceCode: "11"
   name: "北京市"                code: "1101"
                                 name: "北京市"      cityCode: "1101"
                                                     code: "110101"
                                                     name: "东城区"
```

## 二、HTML 结构

```html
<select name="省" id="province"></select>
<select name="市" id="city"></select>
<select name="区" id="area"></select>
<h1>省份-城市-区县</h1>
```

## 三、渲染省份列表

```javascript
var provSelect = document.getElementById("province");
var citySelect = document.getElementById("city");
var areaSelect = document.getElementById("area");

// 循环省份列表，创建option
provinces.forEach(function (item, index) {
    var option = document.createElement("option");
    option.innerText = item.name;
    provSelect.appendChild(option);
});

// 省份渲染完成后，刷新城市列表
cityUpdate();
```

## 四、省份切换 → 更新城市

```javascript
provSelect.onchange = cityUpdate;

function cityUpdate() {
    // 1. 获取当前选中的省份
    var currentProvince = provinces[provSelect.selectedIndex];
    var provinceCode = currentProvince.code;

    // 2. 清空城市列表
    citySelect.innerHTML = "";

    // 3. 遍历城市，筛选当前省份的城市
    cities.forEach(function (item, i) {
        if (item.provinceCode == provinceCode) {
            var option = document.createElement("option");
            option.innerText = item.name;
            citySelect.appendChild(option);
            // 用dataset记录城市编码（关键！）
            option.dataset.index = item.code;
        }
    });

    // 4. 城市列表刷新后，自动刷新区县列表
    areaUpdate();
}
```

### 关键：selectedIndex

```
provinces[provSelect.selectedIndex]

下拉列表的选中索引 与 省份数据数组索引 一一对应
  → 可通过selectedIndex直接获取当前选中的省份数据
```

## 五、城市切换 → 更新区县

```javascript
citySelect.onchange = areaUpdate;

function areaUpdate() {
    // 1. 获取当前选中的城市编码
    //    不能用selectedIndex！因为城市下拉列表的索引和城市数据源索引不对应
    var option = citySelect.selectedOptions[0];
    var cityCode = option.dataset.index;  // 从dataset取出城市编码

    // 2. 清空区县列表
    areaSelect.innerHTML = "";

    // 3. 遍历区县，筛选当前城市的区县
    areas.forEach(function (item, i) {
        if (item.cityCode == cityCode) {
            var option = document.createElement("option");
            option.innerText = item.name;
            areaSelect.appendChild(option);
        }
    });

    // 4. 区县刷新后，更新地址
    locationUpdate();
}
```

### 关键：selectedOptions + dataset

```
城市下拉列表的索引 ≠ 城市数据源索引
  → 不能用 selectedIndex 获取城市数据

解决方案：
  1. 渲染城市option时，用 dataset.index 记录城市编码
  2. 选中时，用 selectedOptions[0] 获取选中的option
  3. 从 option.dataset.index 取出城市编码
```

## 六、更新地址显示

```javascript
areaSelect.onchange = locationUpdate;

function locationUpdate() {
    var provOption = provSelect.selectedOptions[0];
    var cityOption = citySelect.selectedOptions[0];
    var areaOption = areaSelect.selectedOptions[0];

    result.innerText = provOption.innerText + ' - '
                     + cityOption.innerText + ' - '
                     + areaOption.innerText;
}
```

## 七、联动流程

```
页面加载 → 渲染省份 → cityUpdate() → 渲染城市 → areaUpdate() → 渲染区县 → locationUpdate()

省份切换 → cityUpdate() → 重新渲染城市 → areaUpdate() → 重新渲染区县 → locationUpdate()

城市切换 → areaUpdate() → 重新渲染区县 → locationUpdate()

区县切换 → locationUpdate()
```

## 八、关键知识点总结

| 知识点 | 用法 | 说明 |
|--------|------|------|
| `forEach` | `arr.forEach(fn)` | 遍历数组 |
| `createElement` | `document.createElement("option")` | 创建元素 |
| `appendChild` | `select.appendChild(option)` | 添加子元素 |
| `selectedIndex` | `select.selectedIndex` | 获取选中项索引 |
| `selectedOptions` | `select.selectedOptions[0]` | 获取选中的option元素 |
| `dataset` | `option.dataset.index` | 自定义数据属性 |
| `innerHTML = ""` | `select.innerHTML = ""` | 清空列表 |
| `onchange` | `select.onchange = fn` | 下拉切换事件 |
