## switch 练习

### 练习1：简单计算器

输入两个数和运算符（+、-、*、/），用 switch 实现计算：

```javascript
var a = prompt("请输入第一个数") * 1;
var op = prompt("请输入运算符（+ - * /）");
var b = prompt("请输入第二个数") * 1;
var result;

switch (op) {
    case "+": result = a + b; break;
    case "-": result = a - b; break;
    case "*": result = a * b; break;
    case "/": result = a / b; break;
    default: result = "不支持的运算符";
}
alert(a + " " + op + " " + b + " = " + result);
```

### 练习2：判断季节

输入月份（1-12），输出对应季节：

```javascript
var month = prompt("请输入月份（1-12）") * 1;
switch (month) {
    case 3:
    case 4:
    case 5:
        alert("春季"); break;
    case 6:
    case 7:
    case 8:
        alert("夏季"); break;
    case 9:
    case 10:
    case 11:
        alert("秋季"); break;
    case 12:
    case 1:
    case 2:
        alert("冬季"); break;
    default:
        alert("月份输入有误");
}
```

## while 练习

### 练习1：求1到n的和

输入正整数 n，求 1+2+3+...+n 的和：

```javascript
var n = prompt("请输入正整数n") * 1;
var sum = 0;
var i = 1;
while (i <= n) {
    sum += i;
    i++;
}
alert("1到" + n + "的和：" + sum);
```

### 练习2：数字位数

输入一个正整数，输出它的位数：

```javascript
var num = prompt("请输入正整数") * 1;
var count = 0;
var temp = num;
while (temp > 0) {
    temp = Math.floor(temp / 10);
    count++;
}
alert(num + "是" + count + "位数");
```

### 练习3：反转数字

输入一个正整数，输出反转后的数字（如 1234 → 4321）：

```javascript
var num = prompt("请输入正整数") * 1;
var rev = 0;
var temp = num;
while (temp > 0) {
    rev = rev * 10 + temp % 10;
    temp = Math.floor(temp / 10);
}
alert(num + " 反转后为 " + rev);
```

## for 练习

### 练习1：求1到100奇数的和

```javascript
var sum = 0;
for (var i = 1; i <= 100; i += 2) {
    sum += i;
}
console.log("奇数和：" + sum);  // 2500
```

### 练习2：求1到100既能被3整除又能被5整除的数之和

```javascript
var sum = 0;
for (var i = 1; i <= 100; i++) {
    if (i % 3 === 0 && i % 5 === 0) {
        sum += i;
    }
}
console.log("和：" + sum);  // 315 (15+30+45+60+75+90)
```

### 练习3：求n的阶乘

输入 n，求 n! = 1×2×3×...×n：

```javascript
var n = prompt("请输入n") * 1;
var result = 1;
for (var i = 1; i <= n; i++) {
    result *= i;
}
alert(n + "! = " + result);
```

### 练习4：斐波那契数列

输出斐波那契数列的前20项：1, 1, 2, 3, 5, 8, 13...

```javascript
var a = 1, b = 1;
var result = a + " " + b + " ";
for (var i = 3; i <= 20; i++) {
    var c = a + b;
    result += c + " ";
    a = b;
    b = c;
}
console.log(result);
```

> 💡 斐波那契规律：每一项等于前两项之和。

### 练习5：判断质数

输入一个数，判断是否为质数：

```javascript
var num = prompt("请输入正整数") * 1;
var isPrime = true;

if (num < 2) {
    isPrime = false;
} else {
    for (var i = 2; i < num; i++) {
        if (num % i === 0) {
            isPrime = false;
            break;  // 找到一个因子就够了
        }
    }
}

alert(num + (isPrime ? "是质数" : "不是质数"));
```

> 💡 优化：只需要判断到 `Math.sqrt(num)` 即可。

### 练习6：求最大公约数

输入两个正整数，求最大公约数（辗转相除法）：

```javascript
var a = prompt("请输入第一个数") * 1;
var b = prompt("请输入第二个数") * 1;
var m = a, n = b;
while (n !== 0) {
    var temp = m % n;
    m = n;
    n = temp;
}
alert(a + "和" + b + "的最大公约数：" + m);
```

## break / continue 练习

### 练习1：找到第一个水仙花数

水仙花数：一个三位数，各位数字的立方和等于该数本身（如 153 = 1³+5³+3³）：

```javascript
for (var i = 100; i <= 999; i++) {
    var bai = Math.floor(i / 100);
    var shi = Math.floor(i / 10) % 10;
    var ge = i % 10;
    if (bai * bai * bai + shi * shi * shi + ge * ge * ge === i) {
        console.log("第一个水仙花数：" + i);  // 153
        break;
    }
}
```

### 练习2：输出所有水仙花数

```javascript
for (var i = 100; i <= 999; i++) {
    var bai = Math.floor(i / 100);
    var shi = Math.floor(i / 10) % 10;
    var ge = i % 10;
    if (bai * bai * bai + shi * shi * shi + ge * ge * ge === i) {
        console.log(i);  // 153, 370, 371, 407
    }
}
```

### 练习3：跳过3的倍数

输出1-20中不是3的倍数的数：

```javascript
for (var i = 1; i <= 20; i++) {
    if (i % 3 === 0) {
        continue;
    }
    console.log(i);
}
```

### 练习4：输出菱形

```
  *
 ***
*****
 ***
  *
```

```javascript
// 上半部分（3行）
for (var i = 1; i <= 3; i++) {
    var row = "";
    for (var j = 1; j <= 3 - i; j++) row += " ";
    for (var k = 1; k <= 2 * i - 1; k++) row += "*";
    console.log(row);
}
// 下半部分（2行）
for (var i = 2; i >= 1; i--) {
    var row = "";
    for (var j = 1; j <= 3 - i; j++) row += " ";
    for (var k = 1; k <= 2 * i - 1; k++) row += "*";
    console.log(row);
}
```

### 练习5：ATM机模拟

```javascript
var balance = 1000;  // 初始余额
while (true) {
    var choice = prompt(
        "请选择操作：\n1.查询余额\n2.存款\n3.取款\n4.退出"
    );
    switch (choice) {
        case "1":
            alert("当前余额：" + balance);
            break;
        case "2":
            var deposit = prompt("请输入存款金额") * 1;
            balance += deposit;
            alert("存款成功，余额：" + balance);
            break;
        case "3":
            var withdraw = prompt("请输入取款金额") * 1;
            if (withdraw > balance) {
                alert("余额不足！");
            } else {
                balance -= withdraw;
                alert("取款成功，余额：" + balance);
            }
            break;
        case "4":
            alert("感谢使用！");
            // 退出整个循环
            // 由于switch中的break只跳出switch，需要额外标志
            choice = "exit";
            break;
        default:
            alert("无效操作");
    }
    if (choice === "exit") break;
}
```

## 综合练习

### 练习1：猜数字游戏

随机生成1-100的数字，让用户猜，提示大了还是小了，直到猜对：

```javascript
var target = Math.floor(Math.random() * 100) + 1;
var count = 0;
var guess;
do {
    guess = prompt("猜一个1-100的数字") * 1;
    count++;
    if (guess > target) {
        alert("猜大了");
    } else if (guess < target) {
        alert("猜小了");
    } else {
        alert("恭喜猜对了！共猜了" + count + "次");
    }
} while (guess !== target);
```

### 练习2：完数

找出1000以内的所有完数（完数 = 其所有因子之和，如6=1+2+3）：

```javascript
for (var i = 2; i <= 1000; i++) {
    var sum = 0;
    for (var j = 1; j < i; j++) {
        if (i % j === 0) {
            sum += j;
        }
    }
    if (sum === i) {
        console.log(i + "是完数");
    }
}
// 输出：6, 28, 496
```
