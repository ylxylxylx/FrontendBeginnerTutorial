## 练习一：去除字符串空格

**目标**：去除字符串中所有的指定字符（如 'v'）

### 方法一：循环 + replace（每次只替换第一个）

```javascript
var str = 'hvvvvvx';

// replace 每次只能替换第一个匹配项
for (var i = 0; i < str.length; i++) {
    if (str.includes('v') == true) {
        str = str.replace('v', '');
        i--; // 替换后字符串变短，i 需要回退
    }
}
console.log(str); // hx
```

**执行过程**：
```
i=0: str包含v → 替换第一个v → str='hvvvvx', i-- → i=0
i=0: str包含v → 替换第一个v → str='hvvvx',  i-- → i=0
i=0: str包含v → 替换第一个v → str='hvvx',   i-- → i=0
i=0: str包含v → 替换第一个v → str='hvx',    i-- → i=0
i=0: str包含v → 替换第一个v → str='hx',     i-- → i=0
i=0: str不包含v → i++ → i=1
i=1: str不包含v → i++ → i=2
...循环结束
```

### 方法二：遍历拼接

```javascript
function delBlank(str) {
    var newStr = '';
    for (var i = 0; i < str.length; i++) {
        // 如果当前字符不是空格，就拼接到新字符串
        newStr += str[i] == ' ' ? '' : str[i];
    }
    return newStr;
}
```

### 方法三：逐字符替换

```javascript
function ch(x) {
    for (var i = 0; i < String(x).length; i++) {
        // 取出每个字符，如果是v就替换为空
        document.write(String(x)[i].replace('v', ''));
    }
}
ch('hvvvvvx');
```

## 练习二：字符串去重

**目标**：去除字符串中重复的字符

```javascript
var str = 'assaaaffaa';
var res = '';

for (var i = 0; i < str.length; i++) {
    // 如果结果字符串中不包含当前字符，就添加
    if (res.indexOf(str[i]) == -1) {
        res += str[i];
    }
}
console.log(res); // asf
```

**思路**：遍历原字符串，用 `indexOf` 判断结果字符串中是否已包含该字符，不包含才添加。

## 练习三：判断一个字符出现了多少次

**目标**：统计字符串中某个字符出现的次数

### 方法一：遍历计数

```javascript
function charCount(str) {
    var count = 0;
    for (var i = 0; i < str.length; i++) {
        if (str[i].includes('a')) {
            count++;
        }
    }
    return count;
}

console.log(charCount('adsadasdasdadasfadadssaddfa'));
```

### 方法二：split 巧用

```javascript
var str = 'adsadasdasdadasfadadssaddfa';
// 用 'a' 字符分隔字符串得到数组，数组的长度减一就是 'a' 的个数
var arr = str.split('a');
console.log(arr.length - 1);
```

> **原理**：`'aba'.split('b')` → `['a', 'a']`，长度2，b出现1次 = 2-1

## 练习四：闰年判断

**目标**：封装函数判断某年是否是闰年

### 闰年规则

- 能被 400 整除 → 闰年
- 能被 4 整除且不能被 100 整除 → 闰年

```javascript
function isLeepYear(year) {
    if (year % 400 == 0) {
        return true;
    } else {
        if (year % 4 == 0 && year % 100 != 0) {
            return true;
        }
        return false;
    }
}
```

### 使用函数作为判断条件

```javascript
// 方式一：用变量接收返回值
var isLeep = isLeepYear(2000);
if (isLeep) {
    console.log('是闰年');
} else {
    console.log('不是闰年');
}

// 方式二：直接用函数作为条件（返回值是布尔值）
if (isLeepYear(2004)) {
    console.log('是闰年');
}
```

### 批量判断：2000-3000年之间的闰年

```javascript
for (var i = 2000; i < 3001; i++) {
    if (isLeepYear(i)) {
        console.log(i, '是闰年');
    }
}
```

## 练习五：质数判断

**目标**：封装函数判断一个数是否是质数

### 质数定义

大于1的自然数，除了1和它本身外，不能被其他自然数整除。

```javascript
function isPrime(n) {
    if (n < 2) {
        return false; // 小于2的不是质数
    } else {
        for (var i = 2; i < n; i++) { // i 的取值范围 [2, n-1]
            if (n % i == 0) {
                return false; // 能被整除，不是质数
            }
        }
        return true; // 循环结束都没被整除，是质数
    }
}
```

### 找出 0-100 之间的所有质数

```javascript
for (var i = 0; i < 100; i++) {
    if (isPrime(i)) {
        console.log('质数：' + i);
    }
}
```
