## 作业1：删除数组中所有的偶数

给定一个数字数组，删除其中所有的偶数，只保留奇数。

**提示**：
- 遍历数组，判断每个元素是否为偶数（`n % 2 === 0`）
- 可以用 `filter` 方法，也可以用 `splice` 方法（注意 splice 删除时索引的变化）
- 如果用 splice 从后往前删除，可以避免索引错位问题

**答案**：

```javascript
// 方法1：使用 filter（推荐，简洁）
var arr = [1, 2, 3, 4, 5, 6, 7, 8];
var res = arr.filter(function(item) {
    return item % 2 !== 0;  // 保留奇数
});
console.log(res);  // [1, 3, 5, 7]

// 方法2：使用 splice 从后往前删除
var arr = [1, 2, 3, 4, 5, 6, 7, 8];
for (var i = arr.length - 1; i >= 0; i--) {
    if (arr[i] % 2 === 0) {
        arr.splice(i, 1);  // 从后往前删除，不影响前面元素的索引
    }
}
console.log(arr);  // [1, 3, 5, 7]
```

## 作业2：判断数组中所有数字是否都是质数

给定一个数字数组，判断数组中的**所有**数字是否都是质数，返回 `true` 或 `false`。

**提示**：
- 复用练习3中的 `isPrime` 函数
- 可以用 `every` 方法，也可以用 for 循环
- 只要有一个不是质数就返回 `false`

**答案**：

```javascript
// 判断质数的函数
function isPrime(n) {
    if (n < 2) return false;
    for (var i = 2; i < n; i++) {
        if (n % i === 0) return false;
    }
    return true;
}

// 方法1：使用 every
var arr = [2, 3, 5, 7];
var res = arr.every(function(item) {
    return isPrime(item);
});
console.log(res);  // true

var arr2 = [2, 3, 4, 7];
var res2 = arr2.every(function(item) {
    return isPrime(item);
});
console.log(res2);  // false（4不是质数）

// 方法2：使用 for 循环
function allPrime(arr) {
    for (var i = 0; i < arr.length; i++) {
        if (!isPrime(arr[i])) return false;
    }
    return true;
}
console.log(allPrime([2, 3, 5, 7]));   // true
console.log(allPrime([2, 3, 4, 7]));   // false
```

## 作业3：计算数组中数字的平均值

给定一个数字数组，计算所有数字的平均值。

**提示**：
- 平均值 = 总和 / 个数
- 可以用 `reduce` 方法求和，再除以 `arr.length`
- 注意结果可能需要保留小数位数

**答案**：

```javascript
var arr = [10, 20, 30, 40, 50];

// 方法1：使用 reduce
var sum = arr.reduce(function(prev, current) {
    return prev + current;
});
var avg = sum / arr.length;
console.log(avg);  // 30

// 方法2：使用 for 循环
function getAverage(arr) {
    var sum = 0;
    for (var i = 0; i < arr.length; i++) {
        sum += arr[i];
    }
    return sum / arr.length;
}
console.log(getAverage(arr));  // 30

// 保留两位小数
console.log(getAverage([1, 2, 3]).toFixed(2));  // "2.00"
```

## 作业4：数组排重

给定一个数组，去除其中重复的元素，返回一个没有重复值的新数组。

**提示**：
- 方法1：用 `indexOf` 判断新数组中是否已存在该元素
- 方法2：利用对象的属性名不重复的特性
- 方法3：ES6 的 `Set` 对象（`[...new Set(arr)]`）

**答案**：

```javascript
var arr = [1, 2, 3, 2, 4, 3, 5, 1];

// 方法1：使用 indexOf
var newArr = [];
for (var i = 0; i < arr.length; i++) {
    if (newArr.indexOf(arr[i]) === -1) {
        newArr.push(arr[i]);
    }
}
console.log(newArr);  // [1, 2, 3, 4, 5]

// 方法2：利用对象属性名不重复
function unique(arr) {
    var obj = {};
    var newArr = [];
    for (var i = 0; i < arr.length; i++) {
        if (!obj[arr[i]]) {
            obj[arr[i]] = true;
            newArr.push(arr[i]);
        }
    }
    return newArr;
}
console.log(unique(arr));  // [1, 2, 3, 4, 5]

// 方法3：使用 ES6 Set（最简洁）
console.log([...new Set(arr)];  // [1, 2, 3, 4, 5]
```

## 作业5：随机获取四位数字不相同的密码

使用数组，随机生成一个4位数字密码，要求4个数字互不相同，数字范围 [0-9]。

**提示**：
- 用 `Math.random()` 和 `Math.floor()` 生成随机数
- 每生成一个数字，检查是否已经存在于密码中
- 可以用 `indexOf` 或 `includes` 判断是否重复
- 循环直到凑齐4个不重复的数字

**答案**：

```javascript
function getPassword() {
    var pwd = [];
    while (pwd.length < 4) {
        // 生成 0-9 的随机整数
        var num = Math.floor(Math.random() * 10);
        // 判断是否已存在
        if (pwd.indexOf(num) === -1) {
            pwd.push(num);
        }
    }
    return pwd.join('');
}

console.log(getPassword());  // 例如 "3851"（每次不同，且4位不重复）
```

## 作业6：字符串反转

给定一个字符串，将其反转输出。例如 `"hello"` 反转为 `"olleh"`。

**提示**：
- 方法1：将字符串转为数组（`split`），反转数组（`reverse`），再合并为字符串（`join`）
- 方法2：从后往前遍历字符串，逐个字符拼接
- 方法3：用 `charAt` 从末尾开始取字符

**答案**：

```javascript
var str = 'hello';

// 方法1：split + reverse + join（最简洁）
var res1 = str.split('').reverse().join('');
console.log(res1);  // 'olleh'

// 方法2：从后往前遍历拼接
function reverseStr(str) {
    var res = '';
    for (var i = str.length - 1; i >= 0; i--) {
        res += str[i];
    }
    return res;
}
console.log(reverseStr(str));  // 'olleh'

// 方法3：用 charAt 从末尾取字符
function reverseStr2(str) {
    var res = '';
    for (var i = str.length - 1; i >= 0; i--) {
        res += str.charAt(i);
    }
    return res;
}
console.log(reverseStr2(str));  // 'olleh'
