## 作业1：用 for 循环创建 table（6行×8列）

**要求**：使用 `createElement` 和 `appendChild` 动态创建一个 6 行 8 列的表格。

**提示**：

```js
// 1. 创建 table 元素
var table = document.createElement('table');
document.body.appendChild(table);

// 2. 外层循环创建行
for (var i = 0; i < 6; i++) {
    var tr = document.createElement('tr');
    table.appendChild(tr);
    // 3. 内层循环创建列
    for (var j = 0; j < 8; j++) {
        var td = document.createElement('td');
        td.innerHTML = i + '-' + j;  // 可选：显示行列号
        td.style.border = '1px solid #ccc';
        td.style.padding = '5px';
        tr.appendChild(td);
    }
}

// 4. 美化表格
table.style.borderCollapse = 'collapse';
```

## 作业2：用数组创建一个下拉列表

**要求**：将数组数据动态生成 `<select>` 下拉列表。

**提示**：

```js
var cities = ['北京', '上海', '广州', '深圳', '杭州'];

// 1. 创建 select
var select = document.createElement('select');
document.body.appendChild(select);

// 2. 遍历数组创建 option
for (var i = 0; i < cities.length; i++) {
    var option = document.createElement('option');
    option.innerHTML = cities[i];
    option.value = cities[i];
    select.appendChild(option);
}
```

## 作业3：纯 JS 存款收益计算器

**要求**：输入本金、利率、期数，计算存款收益。

**提示**：

```js
// 1. 创建输入框和按钮
var div = document.createElement('div');
document.body.appendChild(div);

div.innerHTML = '<input id="benjin" placeholder="本金">' +
    '<input id="lixi" placeholder="利率">' +
    '<input id="qishu" placeholder="期数">' +
    '<button id="btn">计算</button>' +
    '<p id="result"></p>';

// 2. 点击按钮计算
document.getElementById('btn').onclick = function() {
    var benjin = parseFloat(document.getElementById('benjin').value);
    var lixi = parseFloat(document.getElementById('lixi').value);
    var qishu = parseInt(document.getElementById('qishu').value);

    var res = benjin + lixi * qishu;
    document.getElementById('result').innerHTML = '收益：' + res;
};
```

## 作业4：电子时钟

**要求**：页面上实时显示当前时间，每秒更新。

**提示**：

```js
// 1. 创建显示容器
var div = document.createElement('div');
div.style.fontSize = '48px';
div.style.textAlign = 'center';
document.body.appendChild(div);

// 2. 格式化时间函数
function formatTime(date) {
    var h = date.getHours();
    var m = date.getMinutes();
    var s = date.getSeconds();
    // 补零
    h = h < 10 ? '0' + h : h;
    m = m < 10 ? '0' + m : m;
    s = s < 10 ? '0' + s : s;
    return h + ':' + m + ':' + s;
}

// 3. 先显示一次，避免空白
div.innerHTML = formatTime(new Date());

// 4. 每秒更新
setInterval(function() {
    div.innerHTML = formatTime(new Date());
}, 1000);
```
